  owner. Receipt scanning went with it, to Shopping.

  SOURCE BADGES ARE NOT DECORATION
  Every logged row says where its numbers came from. A verified barcode and an
  AI estimate from a photo are not the same claim, and a screen that renders
  them identically is lying about its own confidence.

  WHAT THE SIGN-OFF PASS CHANGED
  Five sentences on this screen said things the build could not back: search
  hits claimed a barcode, the water goal claimed to move, the plan described a

  seed that holds a loss, and the first-weeks caption named a formula nothing

  window.LKFixtures, and every action is written through window.LKStore, so
  the day survives a reload.

  populated, first-weeks (the target is the starting formula until there are

  /* NOT a stored total. eaten and the three macro `have` figures are added up
     from the day's meals every time the list changes, so a meal that is
     logged, rescaled or deleted cannot leave anything behind. */
  function recount(d) {
    var t = { kcal: 0, pro: 0, carb: 0, fat: 0, fibre: 0, sugar: 0, sat: 0, na: 0 };
    d.meals.forEach(function (m) {
      t.kcal += m.kcal || 0; t.pro += m.pro || 0; t.carb += m.carb || 0; t.fat += m.fat || 0;
      var x = m.key ? xOf(m.key) : null;
      var mult = m.mult || 1;
      if (x) { t.fibre += x.fibre * mult; t.sugar += x.sugar * mult; t.sat += x.sat * mult; t.na += x.na * mult; }
      else if (m.micro) { t.fibre += m.micro.fibre || 0; t.sugar += m.micro.sugar || 0; t.sat += m.micro.sat || 0; t.na += m.micro.na || 0; }
    });
    d.eaten = Math.round(t.kcal);
    d.macros[0].have = Math.round(t.pro);
    d.macros[1].have = Math.round(t.carb);
    d.macros[2].have = Math.round(t.fat);
    d.micro = { fibre: Math.round(t.fibre), sugar: Math.round(t.sugar),
                sat: Math.round(t.sat), na: Math.round(t.na) };
    return d;
  }

    w.forEach(function (p, i) {
      var prev = i ? out[i - 1].ema : p.kg;
      out.push({ date: p.date, kg: p.kg, ema: i ? a * p.kg + (1 - a) * prev : p.kg });
    });
    return out;
  }
  function emaRatePctWk() {
    var e = weightEma();
    if (e.length < 2) return null;
    var b = e[e.length - 1], a = e[e.length - 2];
    var days = (isoMs(b.date) - isoMs(a.date)) / 86400000;
    if (!days) return null;
    return ((b.ema - a.ema) / a.ema * 100) * (7 / days);
  }
  /* The hero's figure, and the one the sign-off pass named: the plain change
     across the last week of the log. The seed runs 64.5 kg to 64.2 kg, which
     is -0.47% a week. The screen used to carry a literal weekPct of 0.34 and
     print it as a GAIN. */
  function weekRatePct() {
    var w = weightSeries();
    if (w.length < 2) return null;
    var b = w[w.length - 1], a = w[w.length - 2];
    var days = (isoMs(b.date) - isoMs(a.date)) / 86400000;
    if (!days) return null;
    return ((b.kg - a.kg) / a.kg * 100) * (7 / days);
  }
  /* Whether the rate is going the way the goal wants. "on target" was printed
     unconditionally under a figure whose sign the screen had not looked at. */
  function rateVerdict(pct) {
    if (pct === null) return '';
    if (isCutting()) return pct < -0.1 ? 'on target' : (pct > 0.1 ? 'gaining while cutting' : 'holding');
    if (goalKey() === 'build') return pct > 0.1 ? 'on target' : (pct < -0.1 ? 'losing while building' : 'holding');
    return Math.abs(pct) < 0.2 ? 'on target' : 'drifting';
  }
